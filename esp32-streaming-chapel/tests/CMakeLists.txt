cmake_minimum_required(VERSION 3.16)
project(streaming_chapel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Threads REQUIRED)
add_subdirectory(../../components/Jet/src Jet)
target_include_directories(Jet PUBLIC ../main/firmware)
add_executable(chapel preview.cpp)
target_include_directories(chapel PRIVATE ../main)
target_compile_definitions(chapel PRIVATE ASSET_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../main/assets.bin")
target_link_libraries(chapel PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(chapel PRIVATE /UNDEBUG)
else()
 target_compile_options(chapel PRIVATE -UNDEBUG)
endif()
add_test(NAME chapel COMMAND chapel)

target_compile_definitions(Jet PRIVATE JET_HIGH_PRECISION_UVS=1)

add_executable(chapel_controls controls.cpp)
target_include_directories(chapel_controls PRIVATE ../main)
if(MSVC)
 target_compile_options(chapel_controls PRIVATE /UNDEBUG)
else()
 target_compile_options(chapel_controls PRIVATE -UNDEBUG)
endif()
add_test(NAME chapel_controls COMMAND chapel_controls)
add_test(NAME chapel_tour COMMAND chapel tour no-cache)

add_test(NAME chapel_nearest COMMAND chapel nearest)

add_test(NAME chapel_bilinear COMMAND chapel bilinear)

add_test(NAME chapel_prefilter COMMAND chapel prefilter-test no-cache)

add_executable(chapel_hot hot.cpp)
target_include_directories(chapel_hot PRIVATE ../main)
target_compile_definitions(chapel_hot PRIVATE ASSET_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../main/assets.bin")
target_link_libraries(chapel_hot PRIVATE Jet Threads::Threads)
if(MSVC)
 target_compile_options(chapel_hot PRIVATE /UNDEBUG)
else()
 target_compile_options(chapel_hot PRIVATE -UNDEBUG)
endif()
add_test(NAME chapel_hot COMMAND chapel_hot)

add_executable(chapel_capture capture.cpp)
target_include_directories(chapel_capture PRIVATE ../main)
target_compile_definitions(chapel_capture PRIVATE ASSET_FILE="${CMAKE_CURRENT_SOURCE_DIR}/../main/assets.bin")
target_link_libraries(chapel_capture PRIVATE Jet Threads::Threads)

foreach(check tiled_textures texture_options scene_texture_queue)
    add_executable(jet_${check} ../../components/Jet/tests/${check}.cpp)
    target_link_libraries(jet_${check} PRIVATE Jet Threads::Threads)
    add_test(NAME jet_${check} COMMAND jet_${check})
endforeach()
foreach(check chapel_capture jet_tiled_textures jet_texture_options jet_scene_texture_queue)
    if(MSVC)
        target_compile_options(${check} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${check} PRIVATE -UNDEBUG)
    endif()
endforeach()
